Test code:

cat mylog.cc 
#include <iostream>
#include <cmath>
#include <complex>
#include <limits>

int main ()
{
	double z = std::numeric_limits<double>::infinity();
	std::complex<double> x (0, z);
	std::complex<double> y (0, 0);

	y = std::log(x);
	std::cout << "x = " << x << " log(x) = " << y << std::endl;

	return 0;
}
